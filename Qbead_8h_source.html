118 void setupIMUTapDetection()
119 {
120 // Turn on the accelerometer
121 // Acc = 416Hz (High-Performance mode)
122 imu.writeRegister(LSM6DS3_ACC_GYRO_CTRL1_XL, LSM6DS3_ACC_GYRO_ODR_XL_416Hz);
123
124 // Optionally, disable gyroscope to save power
125 // imu.writeRegister(LSM6DS3_ACC_GYRO_CTRL2_G, LSM6DS3_ACC_GYRO_ODR_G_POWER_DOWN);
126
127 // Enable tap detection in X,Y,Z:
128 uint8_t TAP_CFG_SETTING = LSM6DS3_ACC_GYRO_TIMER_EN_ENABLED | LSM6DS3_ACC_GYRO_TAP_Z_EN_ENABLED | LSM6DS3_ACC_GYRO_TAP_Y_EN_ENABLED | LSM6DS3_ACC_GYRO_TAP_X_EN_ENABLED;
129 imu.writeRegister(LSM6DS3_ACC_GYRO_TAP_CFG1, TAP_CFG_SETTING);
130
131 // Set shock time window:
132 imu.writeRegister(LSM6DS3_ACC_GYRO_INT_DUR2, LSM6DS3_ACC_GYRO_SHOCK_MASK & 0b11);
133
134 // Set tap threshold:
135 uint8_t thrshold_setting = 8; // number between 0 and 31
136 imu.writeRegister(LSM6DS3_ACC_GYRO_TAP_THS_6D, thrshold_setting);
137
138 // Only do single tap detection. Seems like the naming is incorrect?
139 imu.writeRegister(LSM6DS3_ACC_GYRO_WAKE_UP_THS, LSM6DS3_ACC_GYRO_SINGLE_DOUBLE_TAP_DOUBLE_TAP);
140
141 // Single-tap interrupt driven to pin 1
142 imu.writeRegister(LSM6DS3_ACC_GYRO_MD1_CFG, LSM6DS3_ACC_GYRO_INT1_SINGLE_TAP_ENABLED);
143
144 // Enable low pass filter and set cutoff frequency to datarate/400
145 imu.writeRegister(LSM6DS3_ACC_GYRO_CTRL8_XL, LSM6DS3_ACC_GYRO_LPF2_XL_EN | LSM6DS3_ACC_GYRO_LPF2_XL_CUT_ODR_BY_400);
146
147 // Setup interrupt callback
148 pinMode(PIN_LSM6DS3TR_C_INT1, INPUT);
149 attachInterrupt(digitalPinToInterrupt(PIN_LSM6DS3TR_C_INT1), tap_isr, RISING);
150
151 Serial.println("Enabled IMU interrupt!");
152 }

154 // Should use IMU specific reset pin
155 bool resetIMU(LSM6DS3 &imu) {
156 // CTRL3_C = 0x12, SW_RESET = bit 0
157 uint8_t ctrl3c;
158 if (imu.readRegister(&ctrl3c, LSM6DS3_ACC_GYRO_CTRL3_C) != IMU_SUCCESS) return false;
159 imu.writeRegister(LSM6DS3_ACC_GYRO_CTRL3_C, ctrl3c | 0x01);
160 delay(1); // datasheet: reset completes within ~50us, self-clears
161 return imu.begin() == IMU_SUCCESS;
162 }

347 void setBloch_deg_smooth(float theta, float phi, uint32_t c)
348 {
349 if (!checkThetaAndPhi(theta, phi))
350 return;
351 float theta_section = theta / theta_quant;
352 int theta_int = min(nsections - 1, round(theta_section)); // to avoid precision issues near the end of the range
353 int phi_int = round(phi / phi_quant);
354 phi_int = phi_int > nlegs - 1 ? 0 : phi_int;
355
356 float p = (theta_section - theta_int);
357 int theta_direction = sign(p);
358 p = abs(p);
359 float q = 1 - p;
360 p = p * p;
361 q = q * q;
362
363 uint8_t rc = redch(c);
364 uint8_t gc = greench(c);
365 uint8_t bc = bluech(c);
366
367 setLegPixelColor(phi_int, theta_int, color(q * rc, q * gc, q * bc));
368 setLegPixelColor(phi_int, theta_int + theta_direction, color(p * rc, p * gc, p * bc));
369 }

411 static void tap_isr()
412 {
413 // This function is called when the IMU triggers an interrupt. That is: when a tap is detected!
414 // We have to refer to the singletoninstance of the qbead here,
415 // because the one and only qbead object does not exist when this isr is defined.
416 // Then readout XYZ immediately
417 // We could choose to only readout XYZ when we haven't yet processed the last tap,
418 // but for now, let's just update the position everytime we tap.
419 singletoninstance->tappedrecorded = false;
420 singletoninstance->tapped = true;

423 void readIMU(bool print = true)
424 {
425 rbuffer[0] = imu.readFloatAccelX();
426 rbuffer[1] = imu.readFloatAccelY();
427 rbuffer[2] = imu.readFloatAccelZ();
428 rx = (1 - 2 * sx) * rbuffer[ix];
429 ry = (1 - 2 * sy) * rbuffer[iy];
430 rz = (1 - 2 * sz) * rbuffer[iz];
431
432 float rawmag2 = rx * rx + ry * ry + rz * rz;
433
434 float T_new = micros();
435 float delta = T_new - T_imu;
436 T_imu = T_new;
437 const float T = 100000; // 100 ms // TODO make the filter timeconstant configurable
438 if (delta > 100000)
439 {
440 x = rx;
441 y = ry;
442 z = rz;
443 }
444 else
445 {
446 float d = delta / T;
447 x = d * rx + (1 - d) * x;
448 y = d * ry + (1 - d) * y;
449 z = d * rz + (1 - d) * z;
450 }
451 float mag2 = x * x + y * y + z * z;
452
453 t_acc = theta(x, y, z) * 180 / 3.14159;
454 p_acc = phi(x, y) * 180 / 3.14159;
455 if (p_acc < 0)
456 {
457 p_acc += 360;
458 } // to bring it to [0,360] range
459
460 if (!tappedrecorded && tapped)
461 {
462 tappedrecorded = true;
463 whentapped_buffer[0] = imu.readFloatAccelX();
464 whentapped_buffer[1] = imu.readFloatAccelY();
465 whentapped_buffer[2] = imu.readFloatAccelZ();
466 for (uint16_t conn_hdl = 0; conn_hdl < QB_MAX_PRPH_CONNECTION; conn_hdl++)
467 {
468 if (Bluefruit.connected(conn_hdl) && singletoninstance->blecharacc.notifyEnabled(conn_hdl))
469 {
470 singletoninstance->blechartap.notify(singletoninstance->whentapped_buffer, 3 * sizeof(float));
471 }
472 }
473 }
474
475 if (print)
476 {
477 Serial.print(tappedrecorded);
478 Serial.print("\t");
479 Serial.print(tapped);
480 Serial.print("\t");
481 Serial.print(x);
482 Serial.print("\t");
483 Serial.print(y);
484 Serial.print("\t");
485 Serial.print(z);
486 Serial.print("\t");
487 Serial.print(mag2);
488 Serial.print("\t");
489 Serial.print(rawmag2);
490 Serial.print("\t-1\t1\t");
491 Serial.print(t_acc);
492 Serial.print("\t");
493 Serial.print(p_acc);
494 Serial.print("\t-360\t360\t");
495 Serial.println();
496 }
497
498 rbuffer[0] = x;
499 rbuffer[1] = y;
500 rbuffer[2] = z;
501 blecharacc.write(rbuffer, 3 * sizeof(float));
502
503 for (uint16_t conn_hdl = 0; conn_hdl < QB_MAX_PRPH_CONNECTION; conn_hdl++)
504 {
505 if (Bluefruit.connected(conn_hdl) && blecharacc.notifyEnabled(conn_hdl))
506 {
507 blecharacc.notify(rbuffer, 3 * sizeof(float));
508 }
509 }
510 }

512 void startBLEadv(void)
513 {
514 Serial.println("[INFO]{BLE} Start advertising...");
515 // Advertising packet
516 Bluefruit.Advertising.addFlags(BLE_GAP_ADV_FLAGS_LE_ONLY_GENERAL_DISC_MODE);
517 Bluefruit.Advertising.addTxPower();
518
519 // Include HRM Service UUID
520 Bluefruit.Advertising.addService(bleservice);
521
522 // Secondary Scan Response packet (optional)
523 // Since there is no room for 'Name' in Advertising packet
524 Bluefruit.ScanResponse.addName();
525
526 /* Start Advertising
527 * - Enable auto advertising if disconnected
528 * - Interval: fast mode = 20 ms, slow mode = 152.5 ms
529 * - Timeout for fast mode is 30 seconds
530 * - Start(timeout) with timeout = 0 will advertise forever (until connected)
531 *
532 * For recommended advertising interval
533 * https://developer.apple.com/library/content/qa/qa1931/_index.html
534 */
535 Bluefruit.Advertising.restartOnDisconnect(true);
536 Bluefruit.Advertising.setInterval(32, 244); // in unit of 0.625 ms
537 Bluefruit.Advertising.setFastTimeout(30); // number of seconds in fast mode
538 Bluefruit.Advertising.start(0); // 0 = Don't stop advertising after n seconds
539 }
